## Trigram EM Position

```python
def get_em_position(binary_string):
    """
    Input: 3-char string like '101' (Fire)
    Output: EM position (-4 to +4)
    """
    b = [int(x) for x in binary_string]
    spread = abs(b[0] + b[2] - b[1])
    yang_count = sum(b)
    sign = 1 if yang_count <= 1 else -1
    return sign * (1 + spread)
```

## Gate Classification

```python
def classify_gate(inner_trigram, outer_trigram):
    """
    Input: Two 3-char binary strings
    Output: Gate type
    """
    if inner_trigram == outer_trigram:
        return "STANDING_WAVE"
    
    inner_yang = sum(int(x) for x in inner_trigram)
    outer_yang = sum(int(x) for x in outer_trigram)
    
    inner_domain = "VOID" if inner_yang >= 2 else "MATERIAL"
    outer_domain = "VOID" if outer_yang >= 2 else "MATERIAL"
    
    if inner_domain != outer_domain:
        if inner_domain == "VOID":
            return "CROSS_ZERO_MANIFESTING"
        else:
            return "CROSS_ZERO_DEMATERIALISING"
    else:
        return f"SAME_PHASE_{inner_domain}"
```
