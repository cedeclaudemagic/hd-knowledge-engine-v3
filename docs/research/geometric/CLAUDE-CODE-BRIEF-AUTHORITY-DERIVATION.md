function testAuthorityCompleteness() {
  // Generate all 512 definition patterns
  const patterns = [];
  for (let i = 0; i < 512; i++) {
    const binary = i.toString(2).padStart(9, '0');
    patterns.push({
      head: binary[0] === '1',
      ajna: binary[1] === '1',
      throat: binary[2] === '1',
      g: binary[3] === '1',
      heart: binary[4] === '1',

      root: binary[8] === '1'

  function getAuthority(pattern) {
    if (pattern.solarPlexus) return 'Emotional';
    if (pattern.sacral) return 'Sacral';
    if (pattern.spleen) return 'Splenic';
    if (pattern.heart) return 'Ego';
    if (pattern.g) return 'Self-Projected';
    if (pattern.head || pattern.ajna) return 'Mental';
    // Check if anything else is defined

    return 'Lunar';

  // Count authorities
  const counts = {};
  for (const pattern of patterns) {
    const auth = getAuthority(pattern);
    counts[auth] = (counts[auth] || 0) + 1;
  }
  
  // Verify all 512 are categorised
  const total = Object.values(counts).reduce((a, b) => a + b, 0);
  console.log('Total covered:', total, '/ 512');
  console.log('Authority distribution:', counts);
}
